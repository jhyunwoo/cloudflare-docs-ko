## 3. webhook 핸들러 구축

기타 제품`src/index.ts`:

```typescript
import { getSandbox, proxyToSandbox, type Sandbox } from "@cloudflare/sandbox";
import { Octokit } from "@octokit/rest";
import Anthropic from "@anthropic-ai/sdk";

export { Sandbox } from "@cloudflare/sandbox";

interface Env {
	Sandbox: DurableObjectNamespace<Sandbox>;
	GITHUB_TOKEN: string;
	ANTHROPIC_API_KEY: string;
	WEBHOOK_SECRET: string;
}

export default {
	async fetch(
		request: Request,
		env: Env,
		ctx: ExecutionContext,
	): Promise<Response> {
		const proxyResponse = await proxyToSandbox(request, env);
		if (proxyResponse) return proxyResponse;

		const url = new URL(request.url);

		if (url.pathname === "/webhook" && request.method === "POST") {
			const signature = request.headers.get("x-hub-signature-256");
			const contentType = request.headers.get("content-type") || "";
			const body = await request.text();

			//웹훅 서명 확인
			if (
				!signature ||
				!(await verifySignature(body, signature, env.WEBHOOK_SECRET))
			) {
				return Response.json({ error: "Invalid signature" }, { status: 401 });
			}

			const event = request.headers.get("x-github-event");

			//Parse payload (GitHub는 JSON 또는 form-encoded로 보낼 수 있습니다)
			let payload;
			if (contentType.includes("application/json")) {
				payload = JSON.parse(body);
			} else {
				//핸들 form-encoded 페이로드
				const params = new URLSearchParams(body);
				payload = JSON.parse(params.get("payload") || "{}");
			}

			//개방형 PR
			if (
				event === "pull_request" &&
				(payload.action === "opened" || payload.action === "reopened")
			) {
				console.log(`Starting review for PR #${payload.pull_request.number}`);
				//waitUntil을 사용하여 리뷰가 응답 후 완료됩니다.
				ctx.waitUntil(
					reviewPullRequest(payload, env).catch(console.error),
				);
				return Response.json({ message: "Review started" });
			}

			return Response.json({ message: "Event ignored" });
		}

		return new Response(
			"Code Review Bot\n\nConfigure GitHub webhook to POST /webhook",
		);
	},
};

async function verifySignature(
	payload: string,
	signature: string,
	secret: string,
): Promise<boolean> {
	const encoder = new TextEncoder();
	const key = await crypto.subtle.importKey(
		"raw",
		encoder.encode(secret),
		{ name: "HMAC", hash: "SHA-256" },
		false,
		["sign"],
	);

	const signatureBytes = await crypto.subtle.sign(
		"HMAC",
		key,
		encoder.encode(payload),
	);
	const expected =
		"sha256=" +
		Array.from(new Uint8Array(signatureBytes))
			.map((b) => b.toString(16).padStart(2, "0"))
			.join("");

	return signature === expected;
}

async function reviewPullRequest(payload: any, env: Env): Promise<void> {
	const pr = payload.pull_request;
	const repo = payload.repository;
	const octokit = new Octokit({ auth: env.GITHUB_TOKEN });
	const sandbox = getSandbox(env.Sandbox, `review-${pr.number}`);

	try {
		//게시물 초기 댓글
		console.log("Posting initial comment...");
		await octokit.issues.createComment({
			owner: repo.owner.login,
			repo: repo.name,
			issue_number: pr.number,
			body: "Code review in progress...",
		});
		//Clone 저장소
		console.log("Cloning repository...");
		const cloneUrl = `https://${env.GITHUB_TOKEN}@github.com/${repo.owner.login}/${repo.name}.git`;
		await sandbox.exec(
			`git clone --depth=1 --branch=${pr.head.ref} ${cloneUrl} /workspace/repo`,
		);

		//변경된 파일
		console.log("Fetching changed files...");
		const comparison = await octokit.repos.compareCommits({
			owner: repo.owner.login,
			repo: repo.name,
			base: pr.base.sha,
			head: pr.head.sha,
		});

		const files = [];
		for (const file of (comparison.data.files || []).slice(0, 5)) {
			if (file.status !== "removed") {
				const content = await sandbox.readFile(
					`/workspace/repo/${file.filename}`,
				);
				files.push({
					path: file.filename,
					patch: file.patch || "",
					content: content.content,
				});
			}
		}

		//Claude의 Generate 리뷰
		console.log(`Analyzing ${files.length} files with Claude...`);
		const anthropic = new Anthropic({ apiKey: env.ANTHROPIC_API_KEY });
		const response = await anthropic.messages.create({
			model: "claude-sonnet-4-5",
			max_tokens: 2048,
			messages: [
				{
					role: "user",
					content: `Review this PR:

Title: ${pr.title}

Changed files:
${files.map((f) => `File: ${f.path}\nDiff:\n${f.patch}\n\nContent:\n${f.content.substring(0, 1000)}`).join("\n\n")}

Provide a brief code review focusing on bugs, security, and best practices.`,
				},
			],
		});

		const review =
			response.content[0]?.type === "text"
				? response.content[0].text
				: "No review generated";

		//게시물 리뷰
		console.log("Posting review...");
		await octokit.issues.createComment({
			owner: repo.owner.login,
			repo: repo.name,
			issue_number: pr.number,
			body: `## Code Review\n\n${review}\n\n---\n*Generated by Claude*`,
		});
		console.log("Review complete!");
	} catch (error: any) {
		console.error("Review failed:", error);
		await octokit.issues.createComment({
			owner: repo.owner.login,
			repo: repo.name,
			issue_number: pr.number,
			body: `Review failed: ${error.message}`,
		});
	} finally {
		await sandbox.destroy();
	}
}
```

## 4. 로컬 환경 변수 설정

이름 \*`.dev.vars`로컬 개발을위한 프로젝트 루트의 파일 :

```sh
cat > .dev.vars << EOF
GITHUB_TOKEN=your_github_token_here
ANTHROPIC_API_KEY=your_anthropic_key_here
WEBHOOK_SECRET=your_webhook_secret_here
EOF
```

placeholder 값을 다음과 같이 대체하십시오:

- `GITHUB_TOKEN`: GitHub 개인 액세스 토큰은 repo 권한으로
- `ANTHROPIC_API_KEY`: API 키에서[Anthropic 콘솔](https://console.anthropic.com/)
- `WEBHOOK_SECRET`: 임의 문자열 (예 :`openssl rand -hex 32`)

:::참조
더 보기`.dev.vars`파일은 자동으로 gitignored 및 로컬 개발 중에 사용`npm run dev`.
:::

## 5. Cloudflare 갱도를 가진 국부적으로 서버 노출

로컬에서 실제 GitHub webhooks로 테스트하려면 사용[Cloudflare 갱도](/cloudflare-one/networks/connectors/cloudflare-tunnel/)로컬 개발 서버를 노출합니다.

개발 서버 시작:

```sh
npm run dev
```

별도의 터미널에서 로컬 서버에 터널을 만듭니다.

```sh
cloudflared tunnel --url http://localhost:8787
```

이 URL을 출력한다 (예를 들어,`https://example.trycloudflare.com`). 다음 단계의 URL을 복사합니다.

:::참조
당신이하지 않는 경우`cloudflared`설치, 참조[다운로드](/cloudflare-one/networks/connectors/cloudflare-tunnel/downloads/).
:::

## 6. 로컬 테스트를 위한 GitHub webhook 구성

:::주\[중요]
이 webhook 구성**GitHub 저장소**테스트 풀 요청을 만들 수 있습니다. 봇은 웹훅이 구성되는 저장소에 PR을 검토합니다.
주요 특징

1. GitHub의 테스트 저장소에 Navigate
2. 오시는 길**계정 설정** > **웹훅** > **webhook 추가**
3. 설치하기**페이로드 URL**: Cloudflare 단계 5에서 터널 URL`/webhook`appended (예를 들어,`https://example.trycloudflare.com/webhook`)
4. 설치하기**내용 유형**: `application/json`
5. 설치하기**계정 관리**: 사용 된 동일한 값`WEBHOOK_SECRET`내 계정`.dev.vars`이름 \*
6. 제품정보**개인 이벤트를 선택하자**→ 확인**풀 요청**
7. 이름 \***webhook 추가**

## 7. 국부적으로 테스트 잡아당기기 요구에

테스트 PR 만들기:

```sh
git checkout -b test-review
echo "console.log('test');" > test.js
git add test.js
git commit -m "Add test file"
git push origin test-review
```

GitHub에서 PR을 엽니다. 봇은 몇 초 안에 검토 코멘트를 게시해야합니다.

## 9. 생산을 위한 Update webhook

1. 당신의 repository에 가기**계정 설정** > **웹훅**
2. 기존 웹훅을 클릭
3. (주)**페이로드 URL**배포된 작업자 URL에:`https://code-review-bot.YOUR_SUBDOMAIN.workers.dev/webhook`
4. 이름 \***웹훅 업데이트**

봇은 이제 생산에서 실행되며 모든 새로운 풀 요청을 자동으로 검토합니다.
