사용 방법[캐시 API](/workers/runtime-apis/cache/)R2 개체를 저장하려면 Cloudflare의 캐시.

:::참조

당신은 필요[사용자 정의 도메인 연결](/workers/configuration/routing/custom-domains/)또는[오시는 길](/workers/configuration/routing/routes/)Cache API를 사용하기 위해 작업자에게. Cloudflare QXQX0675XQQ QXQXdashboard 편집기, 놀이터 미리보기에서 Cache API 작업`*.workers.dev`배포는 영향을받지 않습니다.

:::

```js

export default {
  async fetch(request, env, context) {
    try {
      const url = new URL(request.url);

      //캐시 URL에서 캐시 키를 구성
      const cacheKey = new Request(url.toString(), request);
      const cache = caches.default;

      //캐시에서 값을 이미 사용할 수 있는지 확인
      //그렇지 않다면 R2에서 그것을 태워야하며 캐시에 저장해야합니다.
      //미래 접근
      let response = await cache.match(cacheKey);

      if (response) {
        console.log(`Cache hit for: ${request.url}.`);
        return response;
      }

      console.log(
        `Response for request url: ${request.url} not present in cache. Fetching and caching request.`
      );

      //캐시가없는 경우 R2
      const objectKey = url.pathname.slice(1);
      const object = await env.MY_BUCKET.get(objectKey);
      if (object === null) {
        return new Response('Object Not Found', { status: 404 });
      }

      //적절한 객체 헤더를 설정
      const headers = new Headers();
      object.writeHttpMetadata(headers);
      headers.set('etag', object.httpEtag);

      //Cache API는 Cache-Control 헤더를 존중합니다. 10에 s-max-age 조정
      //최대 10 초 동안 캐시에 응답을 제한합니다.
      //응답으로 만든 모든 변경은 캐시 된 값에 반영됩니다.
      headers.append('Cache-Control', 's-maxage=10');

      response = new Response(object.body, {
        headers,
      });

      //cacheKey로 fetched 응답 저장
      //waitUntil을 사용하여 차단없이 응답을 반환 할 수 있습니다.
      //캐시에 쓰기
      context.waitUntil(cache.put(cacheKey, response.clone()));

      return response;
    } catch (e) {
      return new Response('Error thrown ' + e.message);
    }
  },
};
```
