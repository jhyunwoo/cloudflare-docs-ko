Durable Objects와 함께 일할 때, 당신은 튼튼한 목표를 창조하기 위하여 이용된 이름에 접근할 필요가 있을 것입니다`idFromName()`. 이 이름은 일반적으로 튼튼한 개체가 (사용자 ID, 객실 이름, 또는 리소스 식별자와 같은)를 나타내는 의미있는 식별자입니다.

그러나 현재 구현에 제한이 있습니다. 그러나 내구성있는 개체를 만들 수 있지만`.idFromName(name)`, 당신은 직접 이 이름을 통해 튼튼한 목표 안쪽에 접근할 수 없습니다`this.ctx.id.name`.

더 보기`RpcTarget`아래 패턴은 각 메소드 호출로 이름을 자동으로 운반하는 통신 레이어를 생성하여 솔루션을 제공합니다. 이 API를 깨끗하게 유지하면서 튼튼한 개체는 자신의 이름에 액세스 할 수 있습니다.

당신의 필요에 따라 Metadata를 일시적으로 저장할 수 있습니다.`RpcTarget`class, or use 내구재 Object storage to persist the metadata for the Lifetime of the object.

이 예제는 내구성 객체 메타 데이터를 인식하지 않습니다. 그것은 방법을 보여줍니다 :

1. 이름 \*`RpcTarget`한국어
2. 튼튼한 Object metadata (이 예제의 ID) 설정`RpcTarget`한국어
3. metadata를 튼튼한 Object 메소드에 전달하십시오.
4. 자주 묻는 질문`RpcTarget`사용 후 클래스

```ts
import { DurableObject, RpcTarget } from "cloudflare:workers";

//* RpcTarget을 확장하는 RpcDO 클래스 만들기
//* 이 클래스를 사용하여 튼튼한 개체 메타 데이터 설정
//* 내구재 목표 방법에 있는 metadata를 통과하십시오
//* @param mainDo - 주요 튼튼한 목표 종류
//* @param doIdentifier - 튼튼한 목표의 식별자

export class RpcDO extends RpcTarget {
	constructor(
		private mainDo: MyDurableObject,
		private doIdentifier: string,
	) {
		super();
	}

	//* 내구재 목표 방법에 사용자의 이름을 전달하십시오
	//* @param 사용자 이름 - user's name to pass to the 튼튼한 개체 방법

	async computeMessage(userName: string): Promise<string> {
		//튼튼한 개체 메소드를 호출하고 사용자의 이름을 전달하고 튼튼한 개체 식별자
		return this.mainDo.computeMessage(userName, this.doIdentifier);
	}

	//* 튼튼한 개체 식별자를 사용하지 않고 튼튼한 개체 방법을 호출
	//* @param 사용자 이름 - user's name to pass to the 튼튼한 개체 방법

	async simpleGreeting(userName: string) {
		return this.mainDo.simpleGreeting(userName);
	}
}

//* 튼튼한 개체 클래스 만들기
//* RpcDO 클래스를 사용하여 튼튼한 개체 메타 데이터를 설정할 수 있습니다.

export class MyDurableObject extends DurableObject<Env> {
	constructor(ctx: DurableObjectState, env: Env) {
		super(ctx, env);
	}

	//* RpcDO 클래스 초기화
	//* 당신은 여기에서 튼튼한 목표 metadata를 놓을 수 있습니다
	//* RpcDO 클래스의 인스턴스를 반환합니다.
	//* @param doIdentifier - 튼튼한 목표의 식별자

	async setMetaData(doIdentifier: string) {
		return new RpcDO(this, doIdentifier);
	}

	//* 사용자 이름과 DO 식별자를 사용하여 인사 메시지를 컴파일하는 기능
	//* @param 사용자 이름 - 본문내용 바로가기 주메뉴 바로가기
	//* @param doIdentifier - 튼튼한 목표의 식별자

	async computeMessage(
		userName: string,
		doIdentifier: string,
	): Promise<string> {
		console.log({
			userName: userName,
			durableObjectIdentifier: doIdentifier,
		});
		return `Hello, ${userName}! The identifier of this DO is ${doIdentifier}`;
	}

	//* RpcTarget에 있지 않는 기능
	//* 모든 기능은 RpcTarget에 있을 필요가 없습니다

	private async notInRpcTarget() {
		return "This is not in the RpcTarget";
	}

	//* 사용자 이름이 걸리는 기능은 튼튼한 개체 식별자를 사용하지 않습니다.
	//* @param 사용자 이름 - 본문내용 바로가기 주메뉴 바로가기

	async simpleGreeting(userName: string) {
		//RpcTarget에없는 개인 함수 호출
		console.log(this.notInRpcTarget());

		return `Hello, ${userName}! This doesn't use the DO identifier.`;
	}
}

export default {
	async fetch(request, env, ctx): Promise<Response> {
		let id: DurableObjectId = env.MY_DURABLE_OBJECT.idFromName(
			new URL(request.url).pathname,
		);
		let stub = env.MY_DURABLE_OBJECT.get(id);

		//* RpcTarget을 사용하여 내구성 객체 메타 데이터를 설정
		//* 기다리지 않는 것은 이쪽

		const rpcTarget = stub.setMetaData(id.name ?? "default");

		//RpcTarget을 사용하여 튼튼한 개체 방법을 호출합니다.
		//DO 식별자는 RpcTarget에서 전달됩니다
		const greeting = await rpcTarget.computeMessage("world");

		//튼튼한 개체 식별자를 사용하지 않는 튼튼한 개체 방법을 호출
		const simpleGreeting = await rpcTarget.simpleGreeting("world");

		//RpcTarget을 청소하십시오.
		try {
			(await rpcTarget)[Symbol.dispose]?.();
			console.log("RpcTarget cleaned up.");
		} catch (e) {
			console.error({
				message: "RpcTarget could not be cleaned up.",
				error: String(e),
				errorProperties: e,
			});
		}

		return new Response(greeting, { status: 200 });
	},
} satisfies ExportedHandler<Env>;
```

이 예제는 튼튼한 개체 메타 데이터를 주장합니다. 그것은 이전 예와 비슷한 단계를 보여줍니다, 그러나 식별자를 저장하기 위해 튼튼한 개체 저장을 사용, RpcTarget을 통해 그것을 통과해야 제거.

```ts
import { DurableObject, RpcTarget } from "cloudflare:workers";

//* RpcTarget을 확장하는 RpcDO 클래스 만들기
//* 이 클래스를 사용하여 튼튼한 개체 메타 데이터 설정
//* 내구재 목표 방법에 있는 metadata를 통과하십시오
//* @param mainDo - 주요 튼튼한 목표 종류
//* @param doIdentifier - 튼튼한 목표의 식별자

export class RpcDO extends RpcTarget {
	constructor(
		private mainDo: MyDurableObject,
		private doIdentifier: string,
	) {
		super();
	}

	//* 내구재 목표 방법에 사용자의 이름을 전달하십시오
	//* @param 사용자 이름 - user's name to pass to the 튼튼한 개체 방법

	async computeMessage(userName: string): Promise<string> {
		//튼튼한 개체 메소드를 호출하고 사용자의 이름을 전달하고 튼튼한 개체 식별자
		return this.mainDo.computeMessage(userName, this.doIdentifier);
	}

	//* 튼튼한 개체 식별자를 사용하지 않고 튼튼한 개체 방법을 호출
	//* @param 사용자 이름 - user's name to pass to the 튼튼한 개체 방법

	async simpleGreeting(userName: string) {
		return this.mainDo.simpleGreeting(userName);
	}
}

//* 튼튼한 개체 클래스 만들기
//* RpcDO 클래스를 사용하여 튼튼한 개체 메타 데이터를 설정할 수 있습니다.

export class MyDurableObject extends DurableObject<Env> {
	constructor(ctx: DurableObjectState, env: Env) {
		super(ctx, env);
	}

	//* RpcDO 클래스 초기화
	//* 당신은 여기에서 튼튼한 목표 metadata를 놓을 수 있습니다
	//* RpcDO 클래스의 인스턴스를 반환합니다.
	//* @param doIdentifier - 튼튼한 목표의 식별자

	async setMetaData(doIdentifier: string) {
		//튼튼한 개체 식별자를 저장하는 DO 저장
		await this.ctx.storage.put("doIdentifier", doIdentifier);
		return new RpcDO(this, doIdentifier);
	}

	//* 사용자 이름과 DO 식별자를 사용하여 인사 메시지를 컴파일하는 기능
	//* @param 사용자 이름 - 본문내용 바로가기 주메뉴 바로가기

	async computeMessage(userName: string): Promise<string> {
		//DO 식별자를 저장
		const doIdentifier = await this.ctx.storage.get("doIdentifier");
		console.log({
			userName: userName,
			durableObjectIdentifier: doIdentifier,
		});
		return `Hello, ${userName}! The identifier of this DO is ${doIdentifier}`;
	}

	//* RpcTarget에 있지 않는 기능
	//* 모든 기능은 RpcTarget에 있을 필요가 없습니다

	private async notInRpcTarget() {
		return "This is not in the RpcTarget";
	}

	//* 사용자 이름이 걸리는 기능은 튼튼한 개체 식별자를 사용하지 않습니다.
	//* @param 사용자 이름 - 본문내용 바로가기 주메뉴 바로가기

	async simpleGreeting(userName: string) {
		//RpcTarget에없는 개인 함수 호출
		console.log(this.notInRpcTarget());

		return `Hello, ${userName}! This doesn't use the DO identifier.`;
	}
}

export default {
	async fetch(request, env, ctx): Promise<Response> {
		let id: DurableObjectId = env.MY_DURABLE_OBJECT.idFromName(
			new URL(request.url).pathname,
		);
		let stub = env.MY_DURABLE_OBJECT.get(id);

		//* RpcTarget을 사용하여 내구성 객체 메타 데이터를 설정
		//* 기다리지 않는 것은 이쪽

		const rpcTarget = stub.setMetaData(id.name ?? "default");

		//RpcTarget을 사용하여 튼튼한 개체 방법을 호출합니다.
		//DO 식별자는 튼튼한 개체의 저장에 저장됩니다
		const greeting = await rpcTarget.computeMessage("world");

		//튼튼한 개체 식별자를 사용하지 않는 튼튼한 개체 방법을 호출
		const simpleGreeting = await rpcTarget.simpleGreeting("world");

		//RpcTarget을 청소하십시오.
		try {
			(await rpcTarget)[Symbol.dispose]?.();
			console.log("RpcTarget cleaned up.");
		} catch (e) {
			console.error({
				message: "RpcTarget could not be cleaned up.",
				error: String(e),
				errorProperties: e,
			});
		}

		return new Response(greeting, { status: 200 });
	},
} satisfies ExportedHandler<Env>;
```
