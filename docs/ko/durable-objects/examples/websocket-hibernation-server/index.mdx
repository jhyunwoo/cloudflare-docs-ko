이 예제는 비슷합니다.[WebSocket 서버 구축](/durable-objects/examples/websocket-server/)예를 들어 WebSocket Hibernation API를 사용합니다. WebSocket Hibernation API는 Durable ObjectsXQ에 내장 된 WebSocket 서버 애플리케이션에 대해 선호하며 시간이 크게 줄어들기 때문에 WebSocket 애플리케이션과 잘 결합하는 추가 기능을 제공합니다. 더 많은 정보를 원하시면, 참조[웹 소켓을 가진 Durable Objects를 사용하십시오](/durable-objects/best-practices/websockets/).

:::참조

WebSocket Hibernation는 WebSocket 사용 사례를 설명하는 데 사용할 수 없습니다. Hibernation는 튼튼한 개체가 서버 역할을 할 때만 지원됩니다. WebSockets가 필요한 경우, 참조[WebSocket 클라이언트 작성](/workers/examples/websockets/#write-a-websocket-client).

:::

    ```ts
    import { DurableObject } from 'cloudflare:workers';

    //작업자
    export default {
      async fetch(request: Request, env: Env, ctx: ExecutionContext): Promise<Response> {
        if (request.url.endsWith('/websocket')) {
          //WebSocket 업그레이드 요청을 받기를 기대합니다.
          //하나 있다면, 요청을 받아 WebSocket 응답을 반환합니다.
          const upgradeHeader = request.headers.get('Upgrade');
          if (!upgradeHeader || upgradeHeader !== 'websocket') {
            return new Response('Worker expected Upgrade: websocket', {
              status: 426,
            });
          }

          if (request.method !== 'GET') {
            return new Response('Worker expected GET method', {
              status: 400,
            });
          }

          //우리는 연속 이름 'foo'를 제공함으로써 튼튼한 개체 ID를 코딩하는 것은 어렵습니다.
          //이 Worker의 모든 요청은 동일한 튼튼한 Object 인스턴스로 전송됩니다.
          let stub = env.WEBSOCKET_HIBERNATION_SERVER.getByName("foo");

          return stub.fetch(request);
        }

        return new Response(
          `Supported endpoints:
    /websocket: Expects a WebSocket upgrade request`,
          {
            status: 200,
            headers: {
              'Content-Type': 'text/plain',
            },
          }
        );
      },
    };

    //튼튼한 목표
    export class WebSocketHibernationServer extends DurableObject {
      //모든 WebSocket 연결 트랙 유지
      //DO hibernates를 할 때, constructor에서 재구성
      sessions: Map<WebSocket, { [key: string]: string }>;

      constructor(ctx: DurableObjectState, env: Env) {
        super(ctx, env);
        this.sessions = new Map();

        //튼튼한 개체를 건설의 일부로서,
        //WebSockets 및 웹 사이트
        //`sessions` 맵에 다시 배치하십시오.

        //DO에서 모든 WebSocket 연결을 얻으십시오.
        this.ctx.getWebSockets().forEach((ws) => {
          let attachment = ws.deserializeAttachment();
          if (attachment) {
            //이전에 WebSocket에 적용된 상태인 경우
            //연결 상태를 복원하려면 `sessions` 맵에 추가하십시오.
            this.sessions.set(ws, { ...attachment });
          }
        });

        //WebSockets를 깨지지 않는 애플리케이션 레벨 자동 응답을 설정합니다.
        this.ctx.setWebSocketAutoResponse(new WebSocketRequestResponsePair('ping', 'pong'));
      }

      async fetch(request: Request): Promise<Response> {
        //WebSocket 연결의 두 끝을 만듭니다.
        const webSocketPair = new WebSocketPair();
        const [client, server] = Object.values(webSocketPair);

        //`acceptWebSocket()`를 호출하면 이 WebSocket이 종료되기 시작합니다.
        //튼튼한 목표 내의 요청. 그것은 "accepting"의 효과,
        //그리고 WebSocket을 전송하고 메시지를 수신 할 수 있습니다.
        //`ws.accept()`와는 달리, `this.ctx.acceptWebSocket(ws)`는 웹소켓이 Workers Runtime를 알립니다
        //"hibernatable", 그래서 런타임은 메모리에이 튼튼한 개체를 핀 필요가 없습니다
        //연결이 열려 있습니다. Inactivity의 기간 동안, 튼튼한 목표는 evicted 일 수 있습니다
        //메모리에서 WebSocket 연결은 열려있을 것입니다. 나중에 몇 점에서
        //WebSocket은 메시지를 수신하고, 실행 시간은 튼튼한 개체를 다시 만들 것입니다
        //(`constructor`)를 실행하고 적절한 핸들러에 메시지를 전달합니다.
        this.ctx.acceptWebSocket(server);

        //세션에 임의 UUID 생성.
        const id = crypto.randomUUID();

        //WebSocket 연결 및 serialize에 세션 ID를 첨부합니다.
        //튼튼한 개체가 일어나면 연결 상태를 복원해야합니다.
        server.serializeAttachment({ id });

        //WebSocket 연결을 Active 세션 맵에 추가합니다.
        this.sessions.set(server, { id });

        return new Response(null, {
          status: 101,
          webSocket: client,
        });
      }

      async webSocketMessage(ws: WebSocket, message: ArrayBuffer | string) {
        //WebSocket 연결과 관련된 세션을 가져옵니다.
        const session = this.sessions.get(ws)!;

        //클라이언트에서 메시지를 수신하면 서버는 동일한 메시지, 연결 세션 ID,
        //그리고 "[Durable Object]: "prefix
        ws.send(`[Durable Object] message: ${message}, from: ${session.id}, to: the initiating client. Total connections: ${this.sessions.size}`);

        //모든 WebSocket 연결에 메시지 보내기, 연결된 WebSockets를 통해 루프.
        this.sessions.forEach((attachment, connectedWs) => {
          connectedWs.send(`[Durable Object] message: ${message}, from: ${session.id}, to: all clients. Total connections: ${this.sessions.size}`);
        });

        //연결(ws)을 제외한 모든 WebSocket 연결에 메시지 보내기
        //연결된 WebSockets를 통해 연결(ws)을 필터링합니다.
        this.sessions.forEach((attachment, connectedWs) => {
          if (connectedWs !== ws) {
              connectedWs.send(`[Durable Object] message: ${message}, from: ${session.id}, to: all clients except the initiating client. Total connections: ${this.sessions.size}`);
          }
        });
      }

      async webSocketClose(ws: WebSocket, code: number, reason: string, wasClean: boolean) {
        //서버의 호출 닫기()는 WebSocket close Handhake를 완료합니다.
        ws.close(code, reason);
        this.sessions.delete(ws);
      }
    }
    ```

    ```py
    from workers import DurableObject, Response, WorkerEntrypoint
    from js import WebSocketPair, WebSocketRequestResponsePair
    import uuid

    class Session:
      def __init__(self, *, ws):
        self.ws = ws

    #작업자
    class Default(WorkerEntrypoint):
    	async def fetch(self, request):
    		if request.url.endswith('/websocket'):
    			#WebSocket 업그레이드 요청을 받기를 기대합니다.
    			#하나 있다면, 요청을 받아 WebSocket 응답을 반환합니다.
    			upgrade_header = request.headers.get('Upgrade')
    			if not upgrade_header or upgrade_header != 'websocket':
    				return Response('Worker expected Upgrade: websocket', status=426)

    			if request.method != 'GET':
    				return Response('Worker expected GET method', status=400)

    			#우리는 연속 이름 'foo'를 제공함으로써 튼튼한 개체 ID를 코딩하는 것은 어렵습니다.
    			#이 Worker의 모든 요청은 동일한 튼튼한 Object 인스턴스로 전송됩니다.
    			stub = self.env.WEBSOCKET_HIBERNATION_SERVER.getByName("foo")

    			return await stub.fetch(request)

    		return Response(
    			"""Supported endpoints:
    /websocket: Expects a WebSocket upgrade request""",
    			status=200,
    			headers={'Content-Type': 'text/plain'}
    		)

    #튼튼한 목표
    class WebSocketHibernationServer(DurableObject):
    	def __init__(self, ctx, env):
    		super().__init__(ctx, env)
    		#모든 WebSocket 연결의 추적을 유지, 세션 ID에 의해 키
    		#DO hibernates를 할 때, constructor에서 재구성
    		self.sessions = {}

    		#튼튼한 개체를 건설의 일부로서,
    		#WebSockets 및 웹 사이트
    		#`sessions` 맵에 다시 배치하십시오.

    		#DO에서 모든 WebSocket 연결을 얻으십시오.
    		for ws in self.ctx.getWebSockets():
    			attachment = ws.deserializeAttachment()
    			if attachment:
    				#이전에 WebSocket에 적용된 상태인 경우
    				#연결 상태를 복원하려면 `sessions` 맵에 추가하십시오.
    				#세션 ID를 키로 사용합니다.
    				self.sessions[attachment] = Session(ws=ws)

    		#WebSockets를 깨지지 않는 애플리케이션 레벨 자동 응답을 설정합니다.
    		self.ctx.setWebSocketAutoResponse(WebSocketRequestResponsePair.new('ping', 'pong'))

    	async def fetch(self, request):
    		#WebSocket 연결의 두 끝을 만듭니다.
    		client, server = WebSocketPair.new().object_values()

    		#`acceptWebSocket()`를 호출하면 이 WebSocket이 종료되기 시작합니다.
    		#튼튼한 목표 내의 요청. 그것은 "accepting"의 효과,
    		#그리고 WebSocket을 전송하고 메시지를 수신 할 수 있습니다.
    		#`ws.accept()`와는 달리, `this.ctx.acceptWebSocket(ws)`는 웹소켓이 Workers Runtime를 알립니다
    		#"hibernatable", 그래서 런타임은 메모리에이 튼튼한 개체를 핀 필요가 없습니다
    		#연결이 열려 있습니다. Inactivity의 기간 동안, 튼튼한 목표는 evicted 일 수 있습니다
    		#메모리에서 WebSocket 연결은 열려있을 것입니다. 나중에 몇 점에서
    		#WebSocket은 메시지를 수신하고, 실행 시간은 튼튼한 개체를 다시 만들 것입니다
    		#(`constructor`)를 실행하고 적절한 핸들러에 메시지를 전달합니다.
    		self.ctx.acceptWebSocket(server)

    		#세션에 임의 UUID 생성.
    		id = str(uuid.uuid4())

    		#WebSocket 연결 및 serialize에 세션 ID를 첨부합니다.
    		#튼튼한 개체가 일어나면 연결 상태를 복원해야합니다.
    		server.serializeAttachment(id)

    		#WebSocket 연결이 활성화된 세션의 맵에 추가, 세션 ID에 의해 키.
    		self.sessions[id] = Session(ws=server)

    		return Response(None, status=101, web_socket=client)

    	async def webSocketMessage(self, ws, message):
    		#WebSocket 연결과 관련된 세션 ID를 가져옵니다.
    		session_id = ws.deserializeAttachment()

    		#클라이언트에서 메시지를 수신하면 서버는 동일한 메시지, 연결 세션 ID,
    		#그리고 "[Durable Object]: "prefix
    		ws.send(f"[Durable Object] message: {message}, from: {session_id}, to: the initiating client. Total connections: {len(self.sessions)}")

    		#모든 WebSocket 연결에 메시지 보내기, 연결된 WebSockets를 통해 루프.
    		for session in self.sessions.values():
    			session.ws.send(f"[Durable Object] message: {message}, from: {session_id}, to: all clients. Total connections: {len(self.sessions)}")

    		#연결(ws)을 제외한 모든 WebSocket 연결에 메시지 보내기
    		#연결된 WebSockets를 통해 연결(ws)을 필터링합니다.
    		for session in self.sessions.values():
    			if session.ws != ws:
    				session.ws.send(f"[Durable Object] message: {message}, from: {session_id}, to: all clients except the initiating client. Total connections: {len(self.sessions)}")

    	async def webSocketClose(self, ws, code, reason, wasClean):
    		#서버의 호출 닫기()는 WebSocket close Handhake를 완료합니다.
    		ws.close(code, reason)
    		#WebSocket 첨부 파일에서 세션 ID를 받아 세션에서 제거
    		session_id = ws.deserializeAttachment()
    		if session_id:
    			self.sessions.pop(session_id, None)
    ```
