Worker 내에서 서명된 요청을 모두 확인할 수 있습니다.[웹 암호화 API](https://developer.mozilla.org/en-US/docs/Web/API/Crypto/subtle).

다음 노동자는:

- 요청 URL을 시작으로`/generate/`, 교체`/generate/`이름 \*`/`, 그것의 timestamp를 가진 결과를 서명하고, 응답 몸에 있는 가득 차있는 서명한 URL을 돌려보냅니다.

- 다른 모든 요청 URL을 위해 서명된 URL을 확인하고 요청을 허용하십시오.

    ```js
    import { Buffer } from "node:buffer";

    const encoder = new TextEncoder();

    //HMAC 토큰은 몇 초 동안 유효합니다.
    const EXPIRY = 60;

    export default {
    	/** 이름
    * 이름
    * @param {Request} 요청
    * @param {{SECRET DATA: 문자열}}
    * @returns는*/
    	async fetch(request, env) {
    		//비대칭 키로 사용할 수 있는 비밀 데이터가 필요합니다. 이것은 있어야한다
    		//암호화된 비밀로 노동자에 붙어.
    		//https://developers.cloudflare.com/workers/configuration/secrets/에 Refer
    		const secretKeyData = encoder.encode(
    			env.SECRET_DATA ?? "my secret symmetric key",
    		);

    		//'sign'과 'verify' 작업 모두에 대한 CryptoKey로 비밀을 가져 오기
    		const key = await crypto.subtle.importKey(
    			"raw",
    			secretKeyData,
    			{ name: "HMAC", hash: "SHA-256" },
    			false,
    			["sign", "verify"],
    		);

    		const url = new URL(request.url);

    		///generate에 대한 unauthenticated access를 허용하는 데모 작업자입니다.
    		//실제 응용 프로그램에서 당신은 그것을 확인 할 것
    		//사용자가 서명한 URL만 생성할 수 있습니다.
    		if (url.pathname.startsWith("/generate/")) {
    			url.pathname = url.pathname.replace("/generate/", "/");

    			const timestamp = Math.floor(Date.now() / 1000);

    			//이것은 당신이 확인 할 수있는 요청에 대한 모든 데이터를 포함합니다
    			//여기서 우리는 타임스탬프와 패스명 만 서명하지만, 종종 당신이 원하는 것
    			//더 많은 데이터를 포함 (예를 들면, URL hostname 또는 쿼리 매개 변수)
    			const dataToAuthenticate = `${url.pathname}${timestamp}`;

    			const mac = await crypto.subtle.sign(
    				"HMAC",
    				key,
    				encoder.encode(dataToAuthenticate),
    			);

    			//https://developers.cloudflare.com/workers/runtime-apis/nodejs/에 Refer
    			//Node.js API를 사용하여 자세한 내용은 Workers
    			const base64Mac = Buffer.from(mac).toString("base64");

    			url.searchParams.set("verify", `${timestamp}-${base64Mac}`);

    			return new Response(`${url.pathname}${url.search}`);
    			//모든 비 /generate 요청 확인
    		} else {
    			//최소 필요한 쿼리 매개 변수가 있는지 확인하십시오.
    			if (!url.searchParams.has("verify")) {
    				return new Response("Missing query parameter", { status: 403 });
    			}

    			const [timestamp, hmac] = url.searchParams.get("verify").split("-");

    			const assertedTimestamp = Number(timestamp);

    			const dataToAuthenticate = `${url.pathname}${assertedTimestamp}`;

    			const receivedMac = Buffer.from(hmac, "base64");

    			//암호를 사용하십시오.subtle.verify() 타이밍 공격에 대해 보호하십시오. HMACs 사용
    			//비대칭 키, 당신은 암호를 호출하여 이것을 구현할 수 있습니다. Subtle.sign() 그리고
    			//그런 다음 문자열 비교 -- 이것은 insecure, 문자열 비교로
    			//잠재력에 대한 정보를 유출하는 첫 번째 실수에
    			//공격자.
    			const verified = await crypto.subtle.verify(
    				"HMAC",
    				key,
    				receivedMac,
    				encoder.encode(dataToAuthenticate),
    			);

    			if (!verified) {
    				return new Response("Invalid MAC", { status: 403 });
    			}

    			//서명된 요청은 1 분 후에 만료됩니다. 이 값은 특정 사용 사례에 따라야 합니다.
    			if (Date.now() / 1000 > assertedTimestamp + EXPIRY) {
    				return new Response(
    					`URL expired at ${new Date((assertedTimestamp + EXPIRY) * 1000)}`,
    					{ status: 403 },
    				);
    			}
    		}

    		return fetch(new URL(url.pathname, "https://example.com"), request);
    	},
    };
    ```

    ```ts
    import { Buffer } from "node:buffer";
    import { Hono } from "hono";
    import { proxy } from "hono/proxy";

    const encoder = new TextEncoder();

    //HMAC 토큰은 몇 초 동안 유효합니다.
    const EXPIRY = 60;

    interface Env {
    	SECRET_DATA: string;
    }

    const app = new Hono();

    //URL 생성 요청
    app.get("/generate/*", async (c) => {
    	const env = c.env;

    	//비대칭 키로 사용할 수있는 몇 가지 비밀 데이터가 필요합니다.
    	const secretKeyData = encoder.encode(
    		env.SECRET_DATA ?? "my secret symmetric key",
    	);

    	//'sign'과 'verify' 작업 모두에 대한 CryptoKey로 비밀을 가져옵니다
    	const key = await crypto.subtle.importKey(
    		"raw",
    		secretKeyData,
    		{ name: "HMAC", hash: "SHA-256" },
    		false,
    		["sign", "verify"],
    	);

    	//"/generate/" 접두사 "/"를 대체하십시오
    	let pathname = c.req.path.replace("/generate/", "/");

    	const timestamp = Math.floor(Date.now() / 1000);

    	//인증된 데이터: pathname + 타임스탬프
    	const dataToAuthenticate = `${pathname}${timestamp}`;

    	//로그인
    	const mac = await crypto.subtle.sign(
    		"HMAC",
    		key,
    		encoder.encode(dataToAuthenticate),
    	);

    	//base64에 서명을 변환
    	const base64Mac = Buffer.from(mac).toString("base64");

    	//URL로 인증 매개변수 추가
    	url.searchParams.set("verify", `${timestamp}-${base64Mac}`);

    	return c.text(`${pathname}${url.search}`);
    });

    //다른 모든 요청에 대한 핸들 검증
    app.all("*", async (c) => {
    	const env = c.env;
    	const url = c.req.url;

    	//비대칭 키로 사용할 수있는 몇 가지 비밀 데이터가 필요합니다.
    	const secretKeyData = encoder.encode(
    		env.SECRET_DATA ?? "my secret symmetric key",
    	);

    	//'sign'과 'verify' 작업 모두에 대한 CryptoKey로 비밀을 가져옵니다
    	const key = await crypto.subtle.importKey(
    		"raw",
    		secretKeyData,
    		{ name: "HMAC", hash: "SHA-256" },
    		false,
    		["sign", "verify"],
    	);

    	//요청을 확인하면 인증 매개 변수가 있습니다.
    	if (!c.req.query("verify")) {
    		return c.text("Missing query parameter", 403);
    	}

    	//추출 타임스탬프 및 서명
    	const [timestamp, hmac] = c.req.query("verify")!.split("-");
    	const assertedTimestamp = Number(timestamp);

    	//서명해야 하는 데이터를 재구성
    	const dataToAuthenticate = `${c.req.path}${assertedTimestamp}`;

    	//base64 서명을 ArrayBuffer로 변환
    	const receivedMac = Buffer.from(hmac, "base64");

    	//서명 확인
    	const verified = await crypto.subtle.verify(
    		"HMAC",
    		key,
    		receivedMac,
    		encoder.encode(dataToAuthenticate),
    	);

    	//인증이 실패한 경우, 반환 403
    	if (!verified) {
    		return c.text("Invalid MAC", 403);
    	}

    	//서명이 만료된 경우 확인
    	if (Date.now() / 1000 > assertedTimestamp + EXPIRY) {
    		return c.text(
    			`URL expired at ${new Date((assertedTimestamp + EXPIRY) * 1000)}`,
    			403,
    		);
    	}

    	//확인이 전달되면, example.com에 요청을 프록시
    	return proxy(`https://example.com/${c.req.path}`, ...c.req);
    });

    export default app;
    ```

    ```py
    from workers import WorkerEntrypoint
    from pyodide.ffi import to_js as _to_js
    from js import Response, URL, TextEncoder, Buffer, fetch, Object, crypto

    def to_js(x):
        return _to_js(x, dict_converter=Object.fromEntries)

    encoder = TextEncoder.new()

    #HMAC 토큰은 몇 초 동안 유효합니다.
    EXPIRY = 60

    class Default(WorkerEntrypoint):
        async def fetch(self, request):
            #비밀 키 받기
            secret_key_data = encoder.encode(getattr(self.env, "SECRET_DATA", None) or "my secret symmetric key")

            #'sign'과 'verify' 작업 모두에 대한 CryptoKey로 비밀을 가져옵니다
            key = await crypto.subtle.importKey(
                "raw",
                secret_key_data,
                to_js({"name": "HMAC", "hash": "SHA-256"}),
                False,
                ["sign", "verify"]
            )

            url = URL.new(request.url)

            if url.pathname.startswith("/generate/"):
                url.pathname = url.pathname.replace("/generate/", "/", 1)

                timestamp = int(Date.now() / 1000)

                #Data 인증
                data_to_authenticate = f"{url.pathname}{timestamp}"

                #로그인
                mac = await crypto.subtle.sign(
                    "HMAC",
                    key,
                    encoder.encode(data_to_authenticate)
                )

                #base64로 변환
                base64_mac = Buffer.from(mac).toString("base64")

                #인증 매개변수 설정
                url.searchParams.set("verify", f"{timestamp}-{base64_mac}")

                return Response.new(f"{url.pathname}{url.search}")
            else:
                #자주 묻는 질문
                if not "verify" in url.searchParams:
                    return Response.new("Missing query parameter", status=403)

                verify_param = url.searchParams.get("verify")
                timestamp, hmac = verify_param.split("-")

                asserted_timestamp = int(timestamp)

                data_to_authenticate = f"{url.pathname}{asserted_timestamp}"

                received_mac = Buffer.from(hmac, "base64")

                #서명 확인
                verified = await crypto.subtle.verify(
                    "HMAC",
                    key,
                    received_mac,
                    encoder.encode(data_to_authenticate)
                )

                if not verified:
                    return Response.new("Invalid MAC", status=403)

                #자주 묻는 질문
                if Date.now() / 1000 > asserted_timestamp + EXPIRY:
                    expiry_date = Date.new((asserted_timestamp + EXPIRY) * 1000)
                    return Response.new(f"URL expired at {expiry_date}", status=403)

            #확인이 전달되면 example.com에 프록시
            return fetch(URL.new(f"https://example.com{url.pathname}"), request)
    ```

## WAF를 사용하여 서명 된 요청을 유효

서명 요청에 대한 제공 예 코드는 호환됩니다[`is_timed_hmac_valid_v0()`](/ruleset-engine/rules-language/functions/#hmac-validation)규칙 언어 기능. 이것은 Worker 스크립트에 의해 서명된 요청을 확인할 수 있음을 의미합니다.[주문 규칙](/waf/custom-rules/use-cases/configure-token-authentication/#option-2-configure-using-custom-rules).
