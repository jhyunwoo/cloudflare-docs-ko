:::caution\[제작시 주의사항]

예제 코드에는 일반적인 헤더 키와 값이 포함되어 있습니다.`X-Custom-PSK`·`mypresharedkey`. 귀하의 리소스를 보호하려면 코드를 저장하기 전에 Workers 편집기에서 헤더 키와 값을 변경하십시오.

    ```js
    export default {
    	async fetch(request) {
    		/** 이름
    * @param {문자} PRESHARED AUTH HEADER KEY   테크니컬 열쇠를 검사하는 주문 우두머리
    * @param {문자} PRESHARED AUTH HEADER  밸브 하드 코딩 키 값*/
    		const PRESHARED_AUTH_HEADER_KEY = "X-Custom-PSK";
    		const PRESHARED_AUTH_HEADER_VALUE = "mypresharedkey";
    		const psk = request.headers.get(PRESHARED_AUTH_HEADER_KEY);

    		if (psk === PRESHARED_AUTH_HEADER_VALUE) {
    			//정확한 preshared 헤더 키 공급. 원래의 Fetch 요청.
    			return fetch(request);
    		}

    		//잘못된 키 공급. 요청을 거부합니다.
    		return new Response("Sorry, you have supplied an invalid key.", {
    			status: 403,
    		});
    	},
    };
    ```

  <TabItem label="Python" icon="seti:python">
    ```py
    from workers import WorkerEntrypoint, Response, fetch

    class Default(WorkerEntrypoint):
        async def fetch(self, request):
            PRESHARED_AUTH_HEADER_KEY = "X-Custom-PSK"
            PRESHARED_AUTH_HEADER_VALUE = "mypresharedkey"

            psk = request.headers[PRESHARED_AUTH_HEADER_KEY]

            if psk == PRESHARED_AUTH_HEADER_VALUE:
                #정확한 preshared 헤더 키 공급. 원래의 Fetch 요청.
                return fetch(request)

            #잘못된 키 공급. 요청을 거부합니다.
            return Response("Sorry, you have supplied an invalid key.", status=403)
    ```

  <TabItem label="호노" icon="seti:typescript">
    ```ts
    import { Hono } from 'hono';

    const app = new Hono();

    //인증 Midware 추가
    app.use('*', async (c, next) => {
      /** 이름
    * 인증 일정 정의*/
      const PRESHARED_AUTH_HEADER_KEY = "X-Custom-PSK";
      const PRESHARED_AUTH_HEADER_VALUE = "mypresharedkey";

      //요청 헤더에서 사전 공유 키를 가져옵니다.
      const psk = c.req.header(PRESHARED_AUTH_HEADER_KEY);

      if (psk === PRESHARED_AUTH_HEADER_VALUE) {
        //정확한 preshared 헤더 키 공급. 다음 핸들러로 계속.
        await next();
      } else {
        //잘못된 키 공급. 요청을 거부합니다.
        return c.text("Sorry, you have supplied an invalid key.", 403);
      }
    });

    //모든 인증된 요청을 전달하여 전달
    app.all('*', async (c) => {
      return fetch(c.req.raw);
    });

    export default app;
    ```
