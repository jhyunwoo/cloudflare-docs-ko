    export default {
    	async fetch(request) {
    		const HOMEPAGE_URL = "https://tutorial.cloudflareworkers.com/";
    		const PROTECTED_TYPE = "image/";

    		//자주 묻는 질문
    		const response = await fetch(request);

    		//이미지가 있다면, Hotlink Protection을 기반으로 합니다.
    		//Referer 헤더.
    		const referer = request.headers.get("Referer");
    		const contentType = response.headers.get("Content-Type") || "";

    		if (referer && contentType.startsWith(PROTECTED_TYPE)) {
    			//hostnames가 일치하지 않는 경우에, 그것은 hotlink입니다
    			if (new URL(referer).hostname !== new URL(request.url).hostname) {
    				//웹 사이트에 사용자를 다시
    				return Response.redirect(HOMEPAGE_URL, 302);
    			}
    		}

    		//모든 것은 벌금, 응답을 일반적으로 반환합니다.
    		return response;
    	},
    };

    from workers import WorkerEntrypoint, Response, fetch
    from urllib.parse import urlparse

    class Default(WorkerEntrypoint):
        async def fetch(self, request):
            homepage_url = "https://tutorial.cloudflareworkers.com/"
            protected_type = "image/"

            #자주 묻는 질문
            response = await fetch(request)

            #이미지가 있다면, referer header를 기반으로 Hotlink Protection에 참여하십시오.
            referer = request.headers["Referer"]
            content_type = response.headers["Content-Type"] or ""

            if referer and content_type.startswith(protected_type):
                #hostnames가 일치하지 않는 경우에, 그것은 hotlink입니다
                if urlparse(referer).hostname != urlparse(request.url).hostname:
                    #웹 사이트에 사용자를 다시
                    return Response.redirect(homepage_url, 302)

            #모두는 벌금, 응답을 일반적으로 돌려
            return response

    ```ts
    import { Hono } from 'hono';

    const app = new Hono();

    //핫 링크 보호를위한 Middleware
    app.use('*', async (c, next) => {
      const HOMEPAGE_URL = "https://tutorial.cloudflareworkers.com/";
      const PROTECTED_TYPE = "image/";

      //응답을 얻기 위해 다음 핸들러에 계속
      await next();

      //응답이있는 경우, hotlinking에 대한 체크
      if (c.res) {
        //이미지가 있다면, Referer header를 기반으로 Hotlink Protection을 참여하십시오.
        const referer = c.req.header("Referer");
        const contentType = c.res.headers.get("Content-Type") || "";

        if (referer && contentType.startsWith(PROTECTED_TYPE)) {
          //hostnames가 일치하지 않는 경우에, 그것은 hotlink입니다
          if (new URL(referer).hostname !== new URL(c.req.url).hostname) {
            //웹 사이트에 사용자를 다시
            c.res = c.redirect(HOMEPAGE_URL, 302);
          }
        }
      }
    });

    //Default route handler 를 사용하여 요청을 전달합니다.
    app.all('*', async (c) => {
      //자주 묻는 질문
      return fetch(c.req.raw);
    });

    export default app;
    ```
