  <TabItem label="모델 번호: JavaScript" icon="seti:javascript">
    ```js
    import { parse } from "cookie";
    export default {
    	async fetch(request) {
    		//쿠키의 이름
    		const COOKIE_NAME = "__uid";
    		const cookie = parse(request.headers.get("Cookie") || "");
    		if (cookie[COOKIE_NAME] != null) {
    			//쿠키 값과 응답
    			return new Response(cookie[COOKIE_NAME]);
    		}
    		return new Response("No cookie with name: " + COOKIE_NAME);
    	},
    };
    ```

  <TabItem label="모델 번호: Python" icon="seti:python">
    ```py
    from http.cookies import SimpleCookie
    from workers import WorkerEntrypoint, Response

    class Default(WorkerEntrypoint):
        async def fetch(self, request):
            #쿠키의 이름
            cookie_name = "__uid"

            cookies = SimpleCookie(request.headers["Cookie"] or "")

            if cookie_name in cookies:
                #쿠키 값으로 응답
                return Response(cookies[cookie_name].value)

            return Response("No cookie with name: " + cookie_name)
    ```
  </TabItem>

  <TabItem label="호노" icon="seti:typescript">
    ```ts
    import { Hono } from 'hono';
    import { getCookie } from 'hono/cookie';

    const app = new Hono();

    app.get('*', (c) => {
      //쿠키의 이름
      const COOKIE_NAME = "__uid";

      //호노의 쿠키 헬퍼를 사용하여 특정 쿠키 가치를 얻으십시오.
      const cookieValue = getCookie(c, COOKIE_NAME);

      if (cookieValue) {
        //쿠키 값과 응답
        return c.text(cookieValue);
      }

      return c.text("No cookie with name: " + COOKIE_NAME);
    });

    export default app;
    ```
