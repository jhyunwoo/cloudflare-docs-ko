    ```js
    export default {
    	async fetch(request, env, ctx) {
    		async function sha256(message) {
    			//UTF-8로 인코딩
    			const msgBuffer = await new TextEncoder().encode(message);
    			//해시 메시지
    			const hashBuffer = await crypto.subtle.digest("SHA-256", msgBuffer);
    			//바이트를 hex 문자열로 변환
    			return [...new Uint8Array(hashBuffer)]
    				.map((b) => b.toString(16).padStart(2, "0"))
    				.join("");
    		}
    		try {
    			if (request.method.toUpperCase() === "POST") {
    				const body = await request.clone().text();
    				//Hash the request body to use it as a 부분의 캐시 키
    				const hash = await sha256(body);
    				const cacheUrl = new URL(request.url);
    				//몸의 해시를 우선해서 캐시에 URL을 저장하십시오
    				cacheUrl.pathname = "/posts" + cacheUrl.pathname + hash;
    				//캐시 할 수있는 GET 변환
    				const cacheKey = new Request(cacheUrl.toString(), {
    					headers: request.headers,
    					method: "GET",
    				});

    				const cache = caches.default;
    				//캐시 키 찾기
    				let response = await cache.match(cacheKey);
    				//그렇지 않으면, 근원에서 POST 요구에 fetch 응답
    				if (!response) {
    					response = await fetch(request);
    					ctx.waitUntil(cache.put(cacheKey, response.clone()));
    				}
    				return response;
    			}
    			return fetch(request);
    		} catch (e) {
    			return new Response("Error thrown " + e.message);
    		}
    	},
    };
    ```

  <TabItem label="Python" icon="seti:python">
    ```py
    import hashlib
    from workers import WorkerEntrypoint
    from pyodide.ffi import create_proxy
    from js import fetch, URL, Headers, Request, caches

    class Default(WorkerEntrypoint):
        async def fetch(self, request, _, ctx):
            if 'POST' in request.method:
                #Hash the request body to use it as a 부분의 캐시 키
                body = await request.clone().text()
                body_hash = hashlib.sha256(body.encode('UTF-8')).hexdigest()

                #몸의 해시를 우선해서 캐시에 URL을 저장하십시오
                cache_url = URL.new(request.url)
                cache_url.pathname = "/posts" + cache_url.pathname + body_hash

                #캐시 할 수있는 GET 변환
                headers = Headers.new(dict(request.headers).items())
                cache_key = Request.new(cache_url.toString(), method='GET', headers=headers)

                #캐시 키 찾기
                cache = caches.default
                response = await cache.match(cache_key)

                #그렇지 않으면, 근원에서 POST 요구에 fetch 응답
                if response is None:
                    response = await fetch(request)
                    ctx.waitUntil(create_proxy(cache.put(cache_key, response.clone())))

                return response

            return fetch(request)
    ```

    ```ts
    import { Hono } from "hono";
    import { sha256 } from "hono/utils/crypto";

    const app = new Hono();

    //POST 요청을 캐싱하는 Middleware
    app.post("*", async (c) => {
    	try {
    		//요청 몸 받기
    		const body = await c.req.raw.clone().text();

    		//Hash the request body to use it as part of the 캐시 키
    		const hash = await sha256(body);

    		//캐시 URL 생성
    		const cacheUrl = new URL(c.req.url);

    		//몸의 해시를 우선해서 캐시에 URL을 저장하십시오
    		cacheUrl.pathname = "/posts" + cacheUrl.pathname + hash;

    		//캐시 할 수있는 GET 변환
    		const cacheKey = new Request(cacheUrl.toString(), {
    			headers: c.req.raw.headers,
    			method: "GET",
    		});

    		const cache = caches.default;

    		//캐시 키 찾기
    		let response = await cache.match(cacheKey);

    		//캐시가없는 경우, POST 요청에 대한 fetch 응답
    		if (!response) {
    			response = await fetch(c.req.raw);
    			c.executionCtx.waitUntil(cache.put(cacheKey, response.clone()));
    		}

    		return response;
    	} catch (e) {
    		return c.text("Error thrown " + e.message, 500);
    	}
    });

    //다른 모든 HTTP 방법을 처리
    app.all("*", (c) => {
    	return fetch(c.req.raw);
    });

    export default app;
    ```
