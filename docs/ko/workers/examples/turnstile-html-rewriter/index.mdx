    	async fetch(request, env) {

    			.on("head", {
    				element(element) {
    					//이 경우 `append`XQ를 사용하여 새로운 스크립트를 `head` 요소에 추가합니다.
    					element.append(
    						`<script src="https://challenges.cloudflare.com/turnstile/v0/api.js" async defer></script>`,
    						{ html: true },
    					);
    				},
    			})
    			.on("div", {
    				element(element) {
    					//TURNSTILE ATTR NAME의 ID를 가진 요소가 발견되면 십자형 회전식 문 위젯 요소를 추가하십시오.
    					if (element.getAttribute("id") === TURNSTILE_ATTR_NAME) {
    						element.append(
    							`<div class="cf-turnstile" data-sitekey="${SITE_KEY}"></div>`,
    							{ html: true },
    						);
    					}
    				},
    			})
    			.transform(res);
    		return newRes;
    	},
    };

    ```ts
    import { Hono } from "hono";

    interface Env {
    	SITE_KEY: string;
    	SECRET_KEY: string;
    	TURNSTILE_ATTR_NAME?: string;
    }

    const app = new Hono<{ Bindings: Env }>();

    //Turnstile 위젯을 주사하는 Middleware
    app.use("*", async (c, next) => {
    	const SITE_KEY = c.env.SITE_KEY; //Turnstile 환경의 Sitekey
    	const TURNSTILE_ATTR_NAME = c.env.TURNSTILE_ATTR_NAME || "your_id_to_replace"; //대상 요소 ID

    	//원래의 endpoint를 통해 요청을 처리
    	await next();

    	//HTML 응답 만 처리
    	const contentType = c.res.headers.get("content-type");
    	if (!contentType || !contentType.includes("text/html")) {
    		return;
    	}

    	//modifiable을 만드는 응답을 복제
    	const originalResponse = c.res;
    	const responseBody = await originalResponse.text();

    	//HTMLRewriter 인스턴스를 생성하여 HTML을 수정합니다.
    	const rewriter = new HTMLRewriter()
    		//머리에 Turnstile 스크립트 추가
    		.on("head", {
    			element(element) {
    				element.append(
    					`<script src="https://challenges.cloudflare.com/turnstile/v0/api.js" async defer></script>`,
    					{ html: true },
    				);
    			},
    		})
    		//대상 div에 Turnstile 위젯 추가
    		.on("div", {
    			element(element) {
    				if (element.getAttribute("id") === TURNSTILE_ATTR_NAME) {
    					element.append(
    						`<div class="cf-turnstile" data-sitekey="${SITE_KEY}" data-theme="light"></div>`,
    						{ html: true },
    					);
    				}
    			},
    		});

    	//같은 속성으로 새로운 응답을 만듭니다.
    	const modifiedResponse = new Response(responseBody, {
    		status: originalResponse.status,
    		statusText: originalResponse.statusText,
    		headers: originalResponse.headers,
    	});

    	//HTMLRewriter
    	c.res = rewriter.transform(modifiedResponse);
    });

    //Turnstile 유효성 검사를 위한 POST 요청 처리
    app.post("*", async (c) => {
    	const formData = await c.req.formData();
    	const token = formData.get("cf-turnstile-response");
    	const ip = c.req.header("CF-Connecting-IP");

    	//토큰이 없다면 오류를 반환합니다.
    	if (!token) {
    		return c.text("Missing Turnstile token", 400);
    	}

    	//인증 데이터 준비
    	const verifyFormData = new FormData();
    	verifyFormData.append("secret", c.env.SECRET_KEY || "");
    	verifyFormData.append("response", token.toString());
    	if (ip) verifyFormData.append("remoteip", ip);

    	//Turnstile API로 토큰을 검증
    	const verifyResult = await fetch(
    		"https://challenges.cloudflare.com/turnstile/v0/siteverify",
    		{
    			method: "POST",
    			body: verifyFormData,
    		},
    	);

    	const outcome = await verifyResult.json<{ success: boolean }>;

    	//인증이 실패하면 오류를 반환합니다.
    	if (!outcome.success) {
    		return c.text("The provided Turnstile token was not valid!", 401);
    	}

    	//인증이 성공하면, 원래 요청으로 진행
    	//일반적으로 양식 제출 로직을 처리 할 것

    	//이 예제의 경우, 우리는 단지 성공 응답을 보낼 것입니다
    	return c.text("Form submission successful!");
    });

    //을 위한 기본 핸들러 요청
    app.get("*", async (c) => {
    	//원래 내용 (당신은 당신의 실제 콘텐츠 소스로 이것을 대체)
    	return await fetch(c.req.raw);
    });

    export default app;
    ```

    from workers import WorkerEntrypoint
    from pyodide.ffi import create_proxy
    from js import HTMLRewriter, fetch

    class Default(WorkerEntrypoint):
        async def fetch(self, request):
            site_key = self.env.SITE_KEY
            attr_name = self.env.TURNSTILE_ATTR_NAME
            res = await fetch(request)

            class Append:
                def element(self, element):
                    s = '<script src="https://challenges.cloudflare.com/turnstile/v0/api.js" async defer></script>'
                    element.append(s, {"html": True})

            class AppendOnID:
                def __init__(self, name):
                    self.name = name
                def element(self, element):
                    #요소의 `getAttribute` 메소드를 사용하여 `id` 또는 `class`XQ
                    if element.getAttribute("id") == self.name:
                        div = f'<div class="cf-turnstile" data-sitekey="{site_key}" data-theme="light"></div>'
                        element.append(div, { "html": True })

            #예를 들어, `head`, `div`
            head = create_proxy(Append())
            div = create_proxy(AppendOnID(attr_name))
            new_res = HTMLRewriter.new().on("head", head).on("div", div).transform(res)

            return new_res
