    ```js
    export default {
    	async fetch(request, env, ctx) {
    		const cacheUrl = new URL(request.url);

    		//캐시 URL에서 캐시 키를 구성
    		const cacheKey = new Request(cacheUrl.toString(), request);
    		const cache = caches.default;

    		//캐시에서 값을 이미 사용할 수 있는지 확인
    		//그렇지 않은 경우, 당신은 근원에서 그것을 fetch 필요가 있고, 캐시에 저장하십시오
    		let response = await cache.match(cacheKey);

    		if (!response) {
    			console.log(
    				`Response for request url: ${request.url} not present in cache. Fetching and caching request.`,
    			);
    			//캐시가 없다면, 근원에서 그것을 얻습니다
    			response = await fetch(request);

    			//응답 생성기를 사용하여 응답의 모든 필드를 상속
    			response = new Response(response.body, response);

    			//Cache API는 Cache-Control 헤더를 존중합니다. 10에 s-maxage 조정
    			//최대 10 초 동안 캐시에 응답을 제한합니다.

    			//응답으로 만든 모든 변경은 캐시 된 값에 반영됩니다.
    			response.headers.append("Cache-Control", "s-maxage=10");

    			ctx.waitUntil(cache.put(cacheKey, response.clone()));
    		} else {
    			console.log(`Cache hit for: ${request.url}.`);
    		}
    		return response;
    	},
    };
    ```

  <TabItem label="Python" icon="seti:python">
    ```py
    from workers import WorkerEntrypoint
    from pyodide.ffi import create_proxy
    from js import Response, Request, URL, caches, fetch

    class Default(WorkerEntrypoint):
        async def fetch(self, request):
            cache_url = request.url

            #캐시 URL에서 캐시 키를 구성
            cache_key = Request.new(cache_url, request)
            cache = caches.default

            #캐시에서 값을 이미 사용할 수 있는지 확인
            #그렇지 않은 경우, 당신은 근원에서 그것을 fetch 필요가 있고, 캐시에 저장하십시오
            response = await cache.match(cache_key)

            if response is None:
                print(f"Response for request url: {request.url} not present in cache. Fetching and caching request.")
                #캐시가 없다면, 근원에서 그것을 얻습니다
                response = await fetch(request)
                #응답 생성기를 사용하여 응답의 모든 필드를 상속
                response = Response.new(response.body, response)

                #Cache API는 Cache-Control 헤더를 존중합니다. 10에 s-max-age 조정
                #10 초 동안 캐시에 응답을 제한합니다.
                #응답으로 만든 모든 변경은 캐시 된 값에 반영됩니다.
                response.headers.append("Cache-Control", "s-maxage=10")
                self.ctx.waitUntil(create_proxy(cache.put(cache_key, response.clone())))
            else:
                print(f"Cache hit for: {request.url}.")
            return response
    ```

    ```ts
    import { Hono } from "hono";
    import { cache } from "hono/cache";

    const app = new Hono();

    //hono 내장 캐시 헬퍼를 레버리지
    app.get(
    	"*",
    	cache({
    		cacheName: "my-cache",
    		cacheControl: "max-age=3600", //1시간
    	}),
    );

    //캐시가없는 경우 요청을 처리하는 경로 추가
    app.get("*", (c) => {
    	return c.text("Hello from Hono!");
    });

    export default app;
    ```
