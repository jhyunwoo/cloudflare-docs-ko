```js
export default {
	async fetch(request) {
		//새로운 요청을 구성하려면 원래 요청을 복제
		const newRequest = new Request(request);
		//새로운 요청에 대한 Bot 관리 헤더 설정 : https://developers.cloudflare.com/bots/reference/bot-management-variables/#workers-variables
		newRequest.headers.set("bot-score", request.cf.botManagement.score); //봇 점수 (integer)
		newRequest.headers.set(
			"verified-bot",
			request.cf.botManagement.verifiedBot,
		); //검증된 봇(boolean)
		newRequest.headers.set("ja4", request.cf.botManagement.ja4); //JA4 지문 해시 (문자)
		//원산지의 새로운 요청에 대한 응답
		return await fetch(newRequest);
	},
};
```
