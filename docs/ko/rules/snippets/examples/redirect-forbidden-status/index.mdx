```js
export default {
	async fetch(request) {
		//원래 요청을 원산지로 보내십시오.
		const response = await fetch(request);
		//403개의 상태 코드로 응답된 경우 확인
		if (response.status == 403) {
			//그래서,이 URL로 리디렉션
			const destinationURL = "https://example.com";
			//이 상태 코드로
			const statusCode = 301;
			//연락처
			return Response.redirect(destinationURL, statusCode);
		}
		//그렇지 않으면, 기원의 응답
		else {
			return response;
		}
	},
};
```
