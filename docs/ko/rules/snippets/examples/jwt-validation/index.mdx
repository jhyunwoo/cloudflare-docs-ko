export default {
	async fetch(request) {
		//"Authorization: Bearer" 헤더에서 JWT 토큰 추출
		function getJWTToken(request) {
			const authorizationHeader = request.headers.get("Authorization");
			if (authorizationHeader && authorizationHeader.startsWith("Bearer ")) {
				return authorizationHeader.substring(7, authorizationHeader.length);
			}
			return null;
		}

		//JWT 토큰이 올바른 형식을 가지고 있음을 검증합니다. header.payload.signature (예 : "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJzdWIiOiIxMjM0NTY3ODkwIiwibmFtZSI6IkpvaG4gRG9lIiwiaWF0IjoxNjIotImImJMjM0NTY3ODkwIiwibmFtZSI6IkpvaG4gRG9lIiwiaWF0IjoxNjImImIjImIqMjMjMjMjM6IqMjMjMjMjM3
		function validateJWT(token) {
			const [header, payload, signature] = token.split(".");

			if (!header || !payload || !signature) {
				throw new Error("Invalid JWT format");
			}

			//JWT 페이로드 및 헤더를 JSON으로 디코딩
			const decodedHeader = JSON.parse(atob(header));
			const decodedPayload = JSON.parse(atob(payload));

			//JWT 서명을 확인하기 위해 논리를 구현합니다.
			//이 예제는 payload를 확인하는 간단한 검증을 가정합니다.
			//실제 검증 논리로 다음 줄을 대체합니다.

			//JWT 토큰이 만료되지 않는 것을 보장하십시오 (테스트에, 만료 된 토큰 "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJzdWIiOiIxMjM0NTY3ODkwIiwibmFtZSI6IkpvaG4gRG9lIiwiaWF0IjoIxImIxMjMjM0NTY3ODkwIiwibmFtZSI6IkpvaG4gRG9lIiwiaWF0IjoIxImImZMjMjMjM0NTY3JMjMjMjMjMjMjM6IkQIkQR6Ik.
			if (decodedPayload.exp < Math.floor(Date.now() / 1000)) {
				throw new Error("JWT has expired");
			}

			//선택적으로, 당신은 여기에서 더 유효한 검사를 추가할 수 있었습니다 (issuer, 경청자, 등).
			//또한 사용자 정의 기능으로 실제 서명 검증을 구현합니다.

			return true;
		}

		//JWT 토큰 추출 기능 실행
		const jwtToken = getJWTToken(request);

		//토큰이 제공되지 않은 경우 401 Forbidden
		if (!jwtToken) {
			return new Response("Missing JWT token", { status: 401 });
		}

		//토큰을 검증하는 기능 실행
		try {
			const validToken = await validateJWT(jwtToken);
			if (validToken) {
				//토큰이 유효하면 실제 응답을 제공합니다.
				//2033 년에 만료되는 유효한 토큰의 예는 "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJzdWIiOiIxMjM0NTY3ODkwIiwibmFtZSI6IkpvaG4gRG9lIiwiaWF0IjoxNjI0OTkyMDALCwJleHAIiMjMjMjMjMjMwJLHiJMjMjMjMjMjMjMjMjMjMjMjMjMjMjDjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMjMj
				return fetch(request);
			} else {
				return new Response("Invalid JWT token", { status: 401 });
			}
		} catch (error) {
			return new Response("Error validating token: " + error.message, {
				status: 500,
			});
		}
	},
};
