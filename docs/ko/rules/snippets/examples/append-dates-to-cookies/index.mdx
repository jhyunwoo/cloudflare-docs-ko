```js
export default {
	async fetch(request) {
		const response = await fetch(request);

		//더 이상 immutable 인 응답을 복제
		const newResponse = new Response(response.body, response);

		//동적 expiry 시간을 정의합니다. 24시간 * 60m * 60초 * 1000ms = 86,400,000ms
		const expiry = new Date(Date.now() + 7 * 86400000).toUTCString();
		//그룹 변수 정의. "A" 요청 헤더 "userGroup"이 "premium", "B"일 경우.
		const group = request.headers.get("userGroup") == "premium" ? "A" : "B";

		//값을 가진 사용자 정의 헤더를 승인
		newResponse.headers.append(
			"Set-Cookie",
			`testGroup=${group}; Expires=${expiry}; Path=/`,
		);

		return newResponse;
	},
};
```
