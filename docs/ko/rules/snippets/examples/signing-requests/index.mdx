뒤에 오는 Snippet는:

- 요청 URL을 시작으로`/generate/`, 교체`/generate/`이름 \*`/`, 그것의 timestamp를 가진 결과를 서명하고, 응답 몸에 있는 가득 차있는 서명한 URL을 돌려보냅니다.

- 다른 모든 요청 URL을 위해 서명된 URL을 확인하고 요청을 허용하십시오.

```js
export default {
	async fetch(request) {
		const secretKey = "your_secret_key"; //실제 비밀 키로 교체
		const expiration = 60; //몇 초에 만료 시간 (HMAC 토큰이 유효해야 함)

		const encoder = new TextEncoder();

		//HMAC-SHA256 서명을위한 비밀 키를 가져옵니다
		const key = await crypto.subtle.importKey(
			"raw",
			encoder.encode(secretKey),
			{ name: "HMAC", hash: "SHA-256" },
			false,
			["sign", "verify"],
		);

		const url = new URL(request.url);

		//요청 URL이 /generate/로 시작하는 경우 확인
		if (url.pathname.startsWith("/generate/")) {
			///generate/와/를 대체하십시오
			url.pathname = url.pathname.replace("/generate/", "/");

			const currentTimestamp = Math.floor(Date.now() / 1000); //초에 현재 타임스탬프

			//인증된 데이터: pathname과 timestamp 결합
			const dataToAuthenticate = `${url.pathname}${currentTimestamp}`;

			//HMAC-SHA256 데이터 서명
			const signature = await crypto.subtle.sign(
				"HMAC",
				key,
				encoder.encode(dataToAuthenticate),
			);

			//안전한 방식으로 타임스탬프와 HMAC를 인코딩
			const signatureBase64 = btoa(
				String.fromCharCode(...new Uint8Array(signature)),
			);
			const signedData = `${currentTimestamp}-${signatureBase64}`;
			const encodedSignedData = encodeURIComponent(signedData);

			//서명 URL 생성
			const signedURL = `${url}?verify=${encodedSignedData}`;

			//응답 몸에 서명 된 URL을 반환
			return new Response(signedURL, { status: 200 });
		}

		//다른 모든 URL의 경우, 서명된 URL을 확인합니다
		const params = new URLSearchParams(url.search);
		const verifyParam = params.get("verify");

		if (!verifyParam) {
			return new Response("Verification parameter is missing", { status: 403 });
		}

		//디코드 및 분할 확인 매개 변수 타임스탬프 및 HMAC
		const decodedVerifyParam = decodeURIComponent(verifyParam);
		const [timestampStr, receivedMac] = decodedVerifyParam.split("-");

		//Parse 타임스탬프는 유효 번호입니다
		const timestamp = parseInt(timestampStr, 10);
		if (isNaN(timestamp)) {
			return new Response("Invalid timestamp", { status: 403 });
		}

		//자주 묻는 질문
		const currentTimestamp = Math.floor(Date.now() / 1000);
		if (currentTimestamp > timestamp + expiration) {
			return new Response("Signed URL has expired", { status: 403 });
		}

		//URL을 확인하려면 인증 매개 변수를 제거
		params.delete("verify");
		url.search = params.toString();

		//인증에 대한 인증
		const dataToVerify = `${url.pathname}${timestamp}`;

		//HMAC-SHA256 서명 확인
		const isValid = await crypto.subtle.verify(
			"HMAC",
			key,
			new Uint8Array([...atob(receivedMac)].map((char) => char.charCodeAt(0))),
			encoder.encode(dataToVerify),
		);

		if (!isValid) {
			return new Response("Invalid signature", { status: 403 });
		}

		//서명이 유효하다면 계속 요청
		return fetch(request);
	},
};
```
